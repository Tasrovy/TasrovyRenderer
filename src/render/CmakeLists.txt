    # 资源与内存封装
    "${CMAKE_CURRENT_SOURCE_DIR}/VulkanBuffer.h"

    "${CMAKE_CURRENT_SOURCE_DIR}/VulkanImage.h"

    # 描述符管理
    "${CMAKE_CURRENT_SOURCE_DIR}/VulkanDescriptorSetLayout.h"

    "${CMAKE_CURRENT_SOURCE_DIR}/tiny_obj_loader.h"
)

# 2. 定义静态库 TasrovyRender
add_library(TasrovyRender STATIC ${RENDER_SOURCES})

# 3. 设置头文件包含路径
# 因为你的头文件和源文件在同一个目录下，所以包含路径就是当前目录
# PUBLIC 使得：App 层可以直接 #include <VulkanContext.h> (或者通过相对路径)
target_include_directories(TasrovyRender PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")

# 4. 绑定第三方依赖和内部 Core 依赖
target_link_libraries(TasrovyRender
    PUBLIC
        glm::glm                 # 顶点和矩阵运算
        volk::volk               # 动态 Vulkan 加载器
	imgui::imgui
    PRIVATE
        glfw                     # 仅内部 SwapChain 需要创建窗口 Surface
        Vulkan::Vulkan           # Vulkan SDK
)

# 5. 开启 Volk 的动态函数加载宏 (关键：不声明静态 Vulkan 函数原型)
target_compile_definitions(TasrovyRender PUBLIC VK_NO_PROTOTYPES)
